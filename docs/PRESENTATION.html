Case Study 1 — Reentrancy

// ❌ VULNERABLE (before fix) function withdraw() external { uint256 amt = balances[msg.sender]; msg.sender.call{value: amt}(""); // ← attack balances[msg.sender] = 0; // ← too late } // ✅ FIXED (CEI + ReentrancyGuard) function withdraw() external nonReentrant { uint256 amt = balances[msg.sender]; balances[msg.sender] = 0; // ← Effect first (bool ok,) = msg.sender.call{value: amt}(""); if (!ok) revert TransferFailed(); }
